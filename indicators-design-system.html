XAML Стили и примеры

Стили бейджей

<!-- Warning Badge Style -->
<Style x:Key="WarningBadgeStyle" TargetType="Border">
    <Setter Property="Background" Value="#F59E0B"/>
    <Setter Property="CornerRadius" Value="4"/>
    <Setter Property="Padding" Value="4,2"/>
    <Setter Property="MinWidth" Value="18"/>
    <Setter Property="MinHeight" Value="18"/>
    <Setter Property="RenderTransform">
        <Setter.Value>
            <ScaleTransform ScaleX="0" ScaleY="0"/>
        </Setter.Value>
    </Setter>
    <Setter Property="RenderTransformOrigin" Value="0.5,0.5"/>
</Style>

<!-- Loading Badge Style -->
<Style x:Key="LoadingBadgeStyle" TargetType="Border">
    <Setter Property="Background" Value="#3B82F6"/>
    <Setter Property="CornerRadius" Value="10"/>
    <Setter Property="Width" Value="20"/>
    <Setter Property="Height" Value="20"/>
</Style>

Стили индикаторов внутри вкладок

<!-- Thermal Warning Indicator -->
<Style x:Key="ThermalWarningIndicatorStyle" TargetType="Border">

</Style>

</Style>

<Storyboard x:Key="BadgeAppearStoryboard">
    <DoubleAnimation 
        Storyboard.TargetProperty="RenderTransform.ScaleX"
        From="0" To="1"
        Duration="0:0:0.4">
        <DoubleAnimation.EasingFunction>
            <BackEase Amplitude="0.5" EasingMode="EaseOut"/>
        </DoubleAnimation.EasingFunction>
    </DoubleAnimation>
    <DoubleAnimation 
        Storyboard.TargetProperty="RenderTransform.ScaleY"
        From="0" To="1"
        Duration="0:0:0.4">
        <DoubleAnimation.EasingFunction>
            <BackEase Amplitude="0.5" EasingMode="EaseOut"/>
        </DoubleAnimation.EasingFunction>
    </DoubleAnimation>
    <DoubleAnimation 
        Storyboard.TargetProperty="Opacity"
        From="0" To="1"
        Duration="0:0:0.3"/>
</Storyboard>

<!-- Loading Pulse Animation -->
<Storyboard x:Key="LoadingPulseStoryboard" RepeatBehavior="Forever">
    <DoubleAnimation 
        Storyboard.TargetName="PulseRing"
        Storyboard.TargetProperty="(UIElement.RenderTransform).(ScaleTransform.ScaleX)"
        From="1" To="1.5"
        Duration="0:0:1.5"/>
    <DoubleAnimation 
        Storyboard.TargetName="PulseRing"
        Storyboard.TargetProperty="Opacity"
        From="0.6" To="0"
        Duration="0:0:1.5"/>
</Storyboard>

Пример использования 

<!-- Navigation Tab with Badge -->
<TabControl>
    <TabItem>
        <TabItem.Header>
            <StackPanel Orientation="Horizontal" Spacing="6">
                <TextBlock Text="Тепловой расчёт" />
                <Border x:Name="ThermalBadge"
                       Style="{StaticResource WarningBadgeStyle}"
                       Visibility="{Binding NeedsRecalculation, Converter={StaticResource BoolToVisibilityConverter}}">
                    <TextBlock Text="!" 
                              Foreground="White"
                              FontWeight="Bold"
                              FontSize="11"
                              HorizontalAlignment="Center"
                              VerticalAlignment="Center"/>
                </Border>
            </StackPanel>
        </TabItem.Header>
        
        <!-- Tab Content -->
        <Grid>
            <!-- Warning Indicator -->
            <Border Style="{StaticResource ThermalWarningIndicatorStyle}"
                   Visibility="{Binding ShowWarning, Converter={StaticResource BoolToVisibilityConverter}}">
                <Grid>
                    <Grid.ColumnDefinitions>
                        <ColumnDefinition Width="Auto"/>
                        <ColumnDefinition Width="*"/>
                    </Grid.ColumnDefinitions>
                    
                    <Border Background="#F59E0B" 
                           CornerRadius="12"
                           Width="48" Height="48">
                        <Path Data="M12,2 L1,21 H23 M12,6 L12,14 M12,16 L12,18"
                              Stroke="White" StrokeThickness="2"
                              HorizontalAlignment="Center"
                              VerticalAlignment="Center"/>
                    </Border>
                    
                    <StackPanel Grid.Column="1" Margin="16,0,0,0">
                        <TextBlock Text="Требуется пересчёт тепловых потерь"
                                  FontSize="16" FontWeight="SemiBold"/>
                        <TextBlock Text="Параметры системы изменились..."
                                  Foreground="#4B5563"
                                  Margin="0,4,0,12"/>
                        <StackPanel Orientation="Horizontal" Spacing="8">
                            <Button Content="Пересчитать"
                                   Background="#F59E0B"
                                   Foreground="White"
                                   Command="{Binding RecalculateCommand}"/>
                            <Button Content="Позже"
                                   Style="{StaticResource TextButtonStyle}"/>
                        </StackPanel>
                    </StackPanel>
                </Grid>
            </Border>
        </Grid>
    </TabItem>
</TabControl>

Доступность (Accessibility)

Контрастность

  • ! Orange (#F59E0B) на белом: 4.5:1 ✓
  • Blue (#3B82F6) на белом: 4.6:1 ✓
  • Текст на жёлтом фоне: 12:1 ✓

Screen Reader

  • Бейджи имеют aria-label с описанием состояния
  • Индикаторы используют role="alert"
  • Live regions для динамических обновлений
  • Фокус виден на всех интерактивных элементах

Резюме

  • • Плавные анимации 300-400ms
  • • Цветовое кодирование состояний

  • • Действия для пользователя
